"""
Newton-Raphson Method
Iteration:    x(i+1) = x(i) - f(x(i)) / f'(x(i))
Convergence:  | f(x) * f''(x) / [f'(x)]^2 | < 1

Install:  pip install sympy numpy matplotlib
Example:  f(x) = x**3 - 2*x - 5 , x0 = 2
"""

import sympy as sp
import numpy as np
import matplotlib.pyplot as plt


def newton_raphson(expr_str, x0, tol=1e-10, max_iter=100):
    x = sp.symbols("x")
    f_expr = sp.sympify(expr_str)
    df_expr = sp.diff(f_expr, x)          # f'(x)
    d2f_expr = sp.diff(f_expr, x, 2)      # f''(x)

    f = sp.lambdify(x, f_expr, "numpy")
    df = sp.lambdify(x, df_expr, "numpy")
    d2f = sp.lambdify(x, d2f_expr, "numpy")

    xi = float(x0)
    points = [xi]

    print(f"f(x)   = {f_expr}")
    print(f"f'(x)  = {df_expr}")
    print(f"f''(x) = {d2f_expr}")
    print(f"x0 = {xi}\n")

    count = 0
    for i in range(max_iter):
        slope = df(xi)
        if slope == 0:
            print("f'(x) = 0 -> قسمة على صفر. جرّب Xo تانية.")
            break

        # شرط التقارب
        cond = abs(f(xi) * d2f(xi) / slope**2)
        status = "متحقق" if cond < 1 else "غير متحقق"

        # قانون نيوتن-رافسون
        x_next = float(xi - f(xi) / slope)
        count += 1
        points.append(x_next)
        print(f"Iteration {count}: x{i} = {xi:.8f} -> x{i+1} = {x_next:.10f} "
              f"| f(x{i+1}) = {f(x_next):.2e} | شرط التقارب = {cond:.4f} ({status})")

        if abs(x_next - xi) < tol:        # الناتج بقى ثابت
            xi = x_next
            print(f"\nعدد مرات التعويض = {count}")
            print(f"الجذر = {xi:.10f}")
            print(f"قيمة الدالة عند الجذر = {f(xi):.2e}")
            break
        xi = x_next
    else:
        print("وصلنا للحد الأقصى من التكرارات من غير ثبات.")

    return f, df, points


def plot_result(f, df, points):
    lo, hi = min(points), max(points)
    pad = max((hi - lo) * 0.3, 1)
    xs = np.linspace(lo - pad, hi + pad, 600)
    ys = f(xs) * np.ones_like(xs)

    plt.figure(figsize=(9, 6))
    plt.plot(xs, ys, label="y = f(x)", linewidth=2)
    plt.axhline(0, color="black", linewidth=0.8)

    # المماسات والنقاط (زي الرسمة في المحاضرة)
    for i in range(len(points) - 1):
        a, b = points[i], points[i + 1]
        plt.plot([a, a], [0, f(a)], "k:", linewidth=0.8)      # خط رأسي
        plt.plot([a, b], [f(a), 0], "g-", linewidth=0.8)      # المماس
    for i, p in enumerate(points):
        plt.plot(p, f(p), "ro")
        plt.annotate(f"x{i}", (p, f(p)), textcoords="offset points", xytext=(6, 6))

    plt.title("Newton-Raphson Method")
    plt.xlabel("x")
    plt.ylabel("f(x)")
    plt.grid(True)
    plt.legend()
    plt.show()


if __name__ == "__main__":
    expr = input("ادخل الدالة (مثال: x**3 - 2*x - 5): ")
    x0 = float(input("ادخل قيمة Xo: "))
    f, df, pts = newton_raphson(expr, x0)
    plot_result(f, df, pts)
